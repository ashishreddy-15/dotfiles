# Desktop tools: all from official Fedora repos.
# One package per line; '#' starts a comment. Already-installed ones are skipped.
kitty                      # terminal
nautilus                   # file manager
gvfs-mtp                   # Android phones (MTP) in Nautilus: plug in, set USB to File transfer
gvfs-gphoto2               # phones/cameras in PTP (photo transfer) mode
gvfs-afc                   # iPhones/iPads in Nautilus
gvfs-smb                   # Windows/Samba network shares in Nautilus
libmtp                     # MTP library and udev rules (device access for your user)
adw-gtk3-theme             # GTK3 apps in the libadwaita look (light and dark)
grim                       # screenshot
slurp                      # region select (for screenshots)
wl-clipboard               # wl-copy / wl-paste
cliphist                   # clipboard history
brightnessctl              # laptop screen brightness
ddcutil                    # external monitor brightness over DDC/CI (needs i2c access: see system/services.sh)
playerctl                  # media keys
pavucontrol                # audio mixer
loupe                      # image viewer
papers                     # PDF and document viewer
gnome-calculator           # calculator
baobab                     # disk usage analyzer
xournalpp                  # handwritten notes and PDF annotation (pen tablet)
network-manager-applet     # Wi-Fi tray / connection editor
blueman                    # Bluetooth GUI (blueman-manager), tray stays off
xdg-desktop-portal-gtk     # file pickers for GTK apps
zenity                     # folder dialog in Settings > Wallpaper (themed by gtk-4.0/gtk.css)
qt5-qtwayland              # native Wayland for Qt5 apps
qt6-qtwayland              # native Wayland for Qt6 apps
wf-recorder                # screen recording (also used to check UI animations frame by frame)
python3-pillow             # the bar measures how light the wallpaper is (wallpaper/Wallpaper.qml)
